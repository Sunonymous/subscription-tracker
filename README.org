(ns subscription-tracker.db
  (:require [cljs.spec.alpha :as s]))

(def default-account
  "Shape of a single account entity: where a subscription's money comes from."
  {:id nil
   :name ""
   :last-digits ""
   :created-at nil
   :updated-at nil})

(def default-subscription
  "Shape of a single subscription entity. See README.org schema notes."
  {:id nil
   :name ""
   :cost 0
   :frequency :monthly            ;; #{:weekly :monthly :quarterly :yearly :custom}
   :custom-interval-days nil
   :billing-date nil
   :status :active                ;; #{:active :paused :cancelled}
   :account-id nil                ;; references an entry in :accounts
   :start-date nil
   :category nil
   :auto-renew? true
   :url nil
   :color nil
   :notes nil
   :created-at nil
   :updated-at nil})

(def default-db
  {:subscriptions {}              ;; map of id -> subscription entity
   :accounts      {}})            ;; map of id -> account entity

;; -- Spec ------------------------------------------------------------------
;; Fields shared between entities

(s/def ::id string?)
(s/def ::name string?)
(s/def ::notes (s/nilable string?))
(s/def ::created-at (s/nilable string?))
(s/def ::updated-at (s/nilable string?))

;; Account Spec

(s/def ::last-digits string?)

(s/def ::account
  (s/keys :req-un [::id ::name ::last-digits]
          :opt-un [::created-at ::updated-at]))

(s/def ::accounts (s/map-of ::id ::account))

;; Subscription Spec

(s/def ::cost number?)
(s/def ::frequency #{:weekly :monthly :quarterly :yearly :custom})
(s/def ::custom-interval-days (s/nilable pos-int?))
(s/def ::billing-date string?)
(s/def ::status #{:active :paused :cancelled})
(s/def ::account-id (s/nilable string?))
(s/def ::start-date (s/nilable string?))
(s/def ::category (s/nilable string?))
(s/def ::auto-renew? boolean?)
(s/def ::url (s/nilable string?))
(s/def ::color (s/nilable string?))

(s/def ::subscription
  (s/keys :req-un [::id ::name ::cost ::frequency ::billing-date ::status]
          :opt-un [::custom-interval-days ::account-id ::start-date ::category
                   ::auto-renew? ::url ::color ::notes ::created-at ::updated-at]))

(s/def ::subscriptions (s/map-of ::id ::subscription))

;; Whole app-db

(s/def ::db
  (s/keys :req-un [::subscriptions ::accounts]))

(ns subscription-tracker.localstorage
  (:require [cljs.reader :as reader]))

(def ls-key
  "localStorage key under which the whole app's data is persisted."
  "subscription-tracker/app-db")

(defn store-data!
  "Writes `data` into localStorage as an edn string."
  [data]
  (.setItem js/localStorage ls-key (pr-str data)))

(defn load-data
  "Reads previously persisted data from localStorage, if any."
  []
  (when-let [stored (.getItem js/localStorage ls-key)]
    (reader/read-string stored)))

(ns subscription-tracker.events
  (:require
   [re-frame.core :as re-frame]
   [cljs.spec.alpha :as s]
   [subscription-tracker.db :as db]
   [subscription-tracker.localstorage :as localstorage]
   [day8.re-frame.tracing :refer-macros [fn-traced]]))

;; -- Interceptors ------------------------------------------------------------

(defn check-and-throw
  "Throws an exception if `db` doesn't match the given spec."
  [spec db]
  (when-not (s/valid? spec db)
    (throw (ex-info (str "Spec check failed: " (s/explain-str spec db))
                     {:spec spec :db db}))))

(def check-spec-interceptor
  "Validates app-db against ::db/db after every event; throws if invalid."
  (re-frame/after (partial check-and-throw ::db/db)))

(re-frame/reg-cofx
 :local-store
 (fn [cofx _]
   (assoc cofx :local-store (localstorage/load-data))))

(def ->local-store
  "Persists the :subscriptions/:accounts portion of app-db to localStorage after every event."
  (re-frame/after
   (fn [db _]
     (localstorage/store-data! (select-keys db [:subscriptions :accounts])))))

(def mutating-interceptors
  "Applied to every event that makes a significant change to app-db:
   validates the result against spec, then persists it to localStorage."
  [check-spec-interceptor ->local-store])

;; -- Helpers -------------------------------------------------------------

(defn- new-id []
  (str (random-uuid)))

(defn- now-iso []
  (.toISOString (js/Date.)))

;; -- Events --------------------------------------------------------------

(re-frame/reg-event-fx
 ::initialize-db
 [(re-frame/inject-cofx :local-store)
  check-spec-interceptor
  ->local-store]
 (fn-traced [{:keys [local-store]} _]
            {:db (merge db/default-db local-store)}))

(re-frame/reg-event-db
 ::add-subscription
 mutating-interceptors
 (fn-traced [db [_ subscription]]
            (let [id (new-id)
                  now (now-iso)
                  entity (merge db/default-subscription subscription
                                 {:id id :created-at now :updated-at now})]
              (assoc-in db [:subscriptions id] entity))))

(re-frame/reg-event-db
 ::update-subscription
 mutating-interceptors
 (fn-traced [db [_ id changes]]
            (update-in db [:subscriptions id] merge changes {:updated-at (now-iso)})))

(re-frame/reg-event-db
 ::add-account
 mutating-interceptors
 (fn-traced [db [_ account]]
            (let [id (new-id)
                  now (now-iso)
                  entity (merge db/default-account account
                                 {:id id :created-at now :updated-at now})]
              (assoc-in db [:accounts id] entity))))

(re-frame/reg-event-db
 ::update-account
 mutating-interceptors
 (fn-traced [db [_ id changes]]
            (update-in db [:accounts id] merge changes {:updated-at (now-iso)})))

(re-frame/reg-event-db
 ::remove-subscription
 mutating-interceptors
 (fn-traced [db [_ id]]
            (update db :subscriptions dissoc id)))

(re-frame/reg-event-db
 ::remove-account
 mutating-interceptors
 (fn-traced [db [_ id]]
            (-> db
                (update :accounts dissoc id)
                (update :subscriptions
                        (fn [subscriptions]
                          (reduce-kv (fn [m sub-id sub]
                                       (assoc m sub-id
                                              (cond-> sub
                                                (= (:account-id sub) id)
                                                (assoc :account-id nil))))
                                     {} subscriptions))))))

(ns subscription-tracker.subs
  (:require
   [re-frame.core :as re-frame]))

;; -- Layer 2: extractors -----------------------------------------------------

(re-frame/reg-sub
 ::subscriptions
 (fn [db _]
   (:subscriptions db)))

(re-frame/reg-sub
 ::accounts
 (fn [db _]
   (:accounts db)))

;; -- Layer 3: derived ---------------------------------------------------------

(re-frame/reg-sub
 ::subscription-list
 :<- [::subscriptions]
 (fn [subscriptions _]
   (vals subscriptions)))

(re-frame/reg-sub
 ::subscription
 :<- [::subscriptions]
 (fn [subscriptions [_ id]]
   (get subscriptions id)))

(re-frame/reg-sub
 ::account-list
 :<- [::accounts]
 (fn [accounts _]
   (vals accounts)))

(re-frame/reg-sub
 ::account
 :<- [::accounts]
 (fn [accounts [_ id]]
   (get accounts id)))
